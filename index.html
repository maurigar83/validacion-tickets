// ========================================================= // CONFIGURACIÓN // ========================================================= const ID_HOJA = "1zlcizb43ddHuBS8eKsLkG7D0kxS92BZSaLjK0_khdvY"; const NOMBRE_HOJA = "Tickets"; // ========================================================= // VALIDACIÓN DE TICKETS // ========================================================= function doGet(e) { const codigo = (e && e.parameter && e.parameter.codigo) || ""; const scanId = (e && e.parameter && e.parameter.scanId) || ""; if (!codigo) { return crearPantalla( "info", "VALIDACIÓN DE TICKETS", "Abre el lector para escanear un código QR.", "", "" ); } const token = String(codigo).trim(); // ===================================================== // CONTROL DE DOBLE PETICIÓN // ===================================================== // GitHub genera un scanId único para cada lectura. // Si el navegador repite exactamente la misma petición, // recibimos el mismo scanId y devolvemos el resultado válido. // Una nueva lectura genera otro scanId y, si el ticket ya // ingresó, se muestra "TICKET YA UTILIZADO". // El QR nuevo contiene únicamente el Token QR de la columna Q. if (!/^[a-f0-9]{32}$/i.test(token)) { return crearPantalla( "error", "CÓDIGO INVÁLIDO", "El código QR no tiene un token válido.", "", "" ); } const bloqueo = LockService.getScriptLock(); try { bloqueo.waitLock(10000); const archivo = SpreadsheetApp.openById(ID_HOJA); const hoja = archivo.getSheetByName(NOMBRE_HOJA); if (!hoja) { throw new Error( 'No se encontró la hoja "' + NOMBRE_HOJA + '".' ); } // Buscar el token exclusivamente en la columna Q. const ultimaFila = hoja.getLastRow(); let fila = 0; if (ultimaFila >= 2) { const tokens = hoja .getRange(2, 17, ultimaFila - 1, 1) .getDisplayValues(); for (let i = 0; i < tokens.length; i++) { if (String(tokens[i][0]).trim() === token) { fila = i + 2; break; } } } if (!fila) { return crearPantalla( "error", "TICKET NO ENCONTRADO", "No pudimos verificar este código.", "", "" ); } const datos = hoja .getRange(fila, 1, 1, 17) .getValues()[0]; // Segunda comprobación: el token leído debe coincidir exactamente // con el token almacenado en Q para esa fila. if (String(datos[16]).trim() !== token) { return crearPantalla( "error", "TICKET NO ENCONTRADO", "No pudimos verificar este código.", "", "" ); } const nombre = String(datos[2] || "").trim(); const apellido = String(datos[3] || "").trim(); const tipo = String(datos[7] || "").trim(); const estadoIngreso = String(datos[12] || "") .trim() .toUpperCase() .normalize("NFD") .replace(/[\u0300-\u036f]/g, ""); let nombreCompleto = [nombre, apellido] .filter(Boolean) .join(" "); if (!nombreCompleto) { nombreCompleto = "Sin nombre"; } if (estadoIngreso === "INGRESO") { const fechaIngreso = hoja .getRange(fila, 14) .getDisplayValue(); const horaIngreso = hoja .getRange(fila, 15) .getDisplayValue(); // Si esta petición trae el mismo scanId que acaba // de registrar el ingreso, es una repetición de la // misma lectura, no una nueva lectura del QR. if (scanId) { const propiedades = PropertiesService.getScriptProperties(); const clave = "ULTIMA_LECTURA_QR_" + token; const registro = propiedades.getProperty(clave); if (registro) { try { const datosLectura = JSON.parse(registro); const antiguedad = Date.now() - Number(datosLectura.timestamp || 0); if ( datosLectura.scanId === scanId && antiguedad >= 0 && antiguedad <= 10000 ) { return crearPantalla( "valid", "TICKET VÁLIDO", nombreCompleto, tipo, "Ingreso registrado: " + fechaIngreso + " — " + horaIngreso ); } } catch (error) { console.log( "No se pudo leer el registro de la última lectura." ); } } } return crearPantalla( "used", "TICKET YA UTILIZADO", nombreCompleto, tipo, "Ingreso registrado: " + fechaIngreso + " — " + horaIngreso ); } if (estadoIngreso !== "NO INGRESO") { return crearPantalla( "error", "ESTADO NO VÁLIDO", nombreCompleto, tipo, "Este ticket no tiene un estado de ingreso válido." ); } const ahora = new Date(); const zonaHoraria = Session.getScriptTimeZone(); const fecha = Utilities.formatDate( ahora, zonaHoraria, "dd/MM/yyyy" ); const hora = Utilities.formatDate( ahora, zonaHoraria, "HH:mm:ss" ); hoja .getRange(fila, 13) .setValue("Ingresó"); hoja .getRange(fila, 14) .setValue(fecha); hoja .getRange(fila, 15) .setValue(hora); SpreadsheetApp.flush(); // Guardar la identificación de esta lectura. // Se utiliza únicamente para distinguir una repetición // de la misma petición de una nueva lectura del QR. if (scanId) { const propiedades = PropertiesService.getScriptProperties(); propiedades.setProperty( "ULTIMA_LECTURA_QR_" + token, JSON.stringify({ scanId: scanId, timestamp: Date.now() }) ); } return crearPantalla( "valid", "TICKET VÁLIDO", nombreCompleto, tipo, "Ingreso registrado: " + fecha + " — " + hora ); } catch (error) { console.error(error); return crearPantalla( "error", "NO FUE POSIBLE VALIDAR", "Intenta nuevamente en unos segundos.", "", "" ); } finally { try { bloqueo.releaseLock(); } catch (error) { console.log(error); } } } // ========================================================= // CREAR PANTALLA DE RESULTADO // ========================================================= function crearPantalla( tipo, titulo, nombre, tipoTicket, detalle ) { const estilos = { valid: { color: "#0f7a3f", fondo: "#e9f8ef", icono: "✓" }, used: { color: "#b42318", fondo: "#fff0ef", icono: "!" }, error: { color: "#b42318", fondo: "#fff0ef", icono: "!" }, info: { color: "#1d4ed8", fondo: "#eff6ff", icono: "i" } }; const estado = estilos[tipo] || estilos.info; const datosTicket = nombre ? `
${escaparHtml(nombre)}
${ tipoTicket ? `
${escaparHtml(tipoTicket)}
` : "" } ${ detalle ? `
${escaparHtml(detalle)}
` : "" }
` : ""; const html = ` ${escaparHtml(titulo)} ${ tipo !== "info" ? ` ` : "" }
${estado.icono}

${escaparHtml(titulo)}

${datosTicket} ${ tipo !== "info" ? `
Volviendo al lector en 3 segundos...
` : "" }
`; return HtmlService .createHtmlOutput(html) .setTitle(titulo) .setXFrameOptionsMode( HtmlService.XFrameOptionsMode.ALLOWALL ); } // ========================================================= // ESCAPAR HTML // ========================================================= function escaparHtml(texto) { return String(texto) .replace(/&/g, "&") .replace(//g, ">") .replace(/"/g, """) .replace(/'/g, "'"); } // ========================================================= // BLOQUEAR EDICIÓN DE DATOS DE VENTA // ========================================================= function bloquearVentaAlCompletar(e) { if (!e || !e.range) { return; } const hoja = e.range.getSheet(); if (hoja.getName() !== "Tickets") { return; } const primeraFila = Math.max(2, e.range.getRow()); const ultimaFila = e.range.getRow() + e.range.getNumRows() - 1; for ( let fila = primeraFila; fila <= ultimaFila; fila++ ) { const datosVenta = hoja .getRange(fila, 3, 1, 9) .getDisplayValues()[0]; const nombre = datosVenta[0].trim(); const apellido = datosVenta[1].trim(); const cedula = datosVenta[2].trim(); const telefono = datosVenta[3].trim(); const correo = datosVenta[4].trim(); const tipo = datosVenta[5] .trim() .toUpperCase() .normalize("NFD") .replace(/[\u0300-\u036f]/g, ""); const referencia = datosVenta[7].trim(); const fechaPago = datosVenta[8].trim(); const datosPersonalesCompletos = nombre !== "" && apellido !== "" && cedula !== "" && telefono !== "" && correo !== ""; if (!datosPersonalesCompletos) { continue; } let ventaCompleta = false; if (tipo === "CORTESIA") { ventaCompleta = true; } else if (tipo === "GENERAL") { ventaCompleta = referencia !== "" && fechaPago !== ""; } else { continue; } if (!ventaCompleta) { continue; } protegerRango( hoja, hoja.getRange(fila, 3, 1, 5), "Venta bloqueada — datos — fila " + fila ); protegerRango( hoja, hoja.getRange(fila, 10, 1, 2), "Venta bloqueada — pago — fila " + fila ); } } // ========================================================= // PROTEGER RANGO PARA EL ADMINISTRADOR // ========================================================= function protegerRango( hoja, rango, descripcion ) { const yaProtegido = hoja .getProtections( SpreadsheetApp.ProtectionType.RANGE ) .some(proteccion => { const rangoProtegido = proteccion.getRange(); return ( rangoProtegido.getA1Notation() === rango.getA1Notation() ); }); if (yaProtegido) { return; } const proteccion = rango .protect() .setDescription(descripcion); const administrador = Session.getEffectiveUser(); proteccion.addEditor(administrador); if (proteccion.canDomainEdit()) { proteccion.setDomainEdit(false); } const editores = proteccion.getEditors(); const correoAdministrador = administrador.getEmail(); editores.forEach(editor => { if ( editor.getEmail() !== correoAdministrador ) { try { proteccion.removeEditor(editor); } catch (error) { console.log( "No se pudo retirar editor: " + editor.getEmail() ); } } }); }
